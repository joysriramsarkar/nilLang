### ১. ডেসিমাল এবং মানির পূর্ণাঙ্গ গাণিতিক লজিক (Nilang Frontend)
এই কোডটি আপনার `std/math/decimal.nil` ফাইলে থাকবে। এখানে স্কেল নরমালাইজেশন, যোগ, বিয়োগ, গুণ এবং ভাগের লজিক দেওয়া হলো।

```nilang
// std/math/decimal.nil
namespace math {
    struct Decimal {
        let value: int;    // মেজর ভ্যালু (যেমন: 12345)
        let scale: int;    // দশমিকের ঘর (যেমন: 2 মানে 123.45)
    }

    impl Decimal {
        fn new(value: int, scale: int) -> Decimal {
            return Decimal { value: value, scale: scale };
        }

        // দুটি ডেসিমালের স্কেল সমান করার হেল্পার ফাংশন
        fn align_scales(a: Decimal, b: Decimal) -> (int, int, int) {
            let max_scale = if a.scale > b.scale { a.scale } else { b.scale };
            let a_val = a.value;
            let b_val = b.value;
            
            // স্কেলের পার্থক্য অনুযায়ী 10 দিয়ে গুণ করা
            let diff_a = max_scale - a.scale;
            for i in range(0, diff_a) { a_val = a_val * 10; }
            
            let diff_b = max_scale - b.scale;
            for i in range(0, diff_b) { b_val = b_val * 10; }
            
            return (a_val, b_val, max_scale);
        }

        fn add(self, other: Decimal) -> Decimal {
            let (a, b, scale) = Decimal.align_scales(self, other);
            return Decimal { value: a + b, scale: scale };
        }

        fn sub(self, other: Decimal) -> Decimal {
            let (a, b, scale) = Decimal.align_scales(self, other);
            return Decimal { value: a - b, scale: scale };
        }

        fn mul(self, other: Decimal) -> Decimal {
            // গুণের সময় স্কেল যোগ হয় (যেমন: 2.5 * 2.5 = 6.25, স্কেল 1+1=2)
            return Decimal { value: self.value * other.value, scale: self.scale + other.scale };
        }

        fn div(self, other: Decimal, precision: int) -> core.Result<Decimal, string> {
            if other.value == 0 {
                return core.Err("Division by zero");
            }
            // ভাগের ক্ষেত্রে প্রিসিশন ধরে রাখতে 10^precision দিয়ে গুণ করা হয়
            let multiplier = 1;
            for i in range(0, precision) { multiplier = multiplier * 10; }
            
            let new_val = (self.value * multiplier) / other.value;
            let new_scale = self.scale - other.scale + precision;
            
            return core.Ok(Decimal { value: new_val, scale: new_scale });
        }
    }

    struct Money {
        let amount: Decimal;
        let currency: string; // "BDT"
    }

    impl Money {
        fn add(self, other: Money) -> core.Result<Money, string> {
            if self.currency != other.currency {
                return core.Err("Currency mismatch: \(self.currency) vs \(other.currency)");
            }
            return core.Ok(Money { amount: self.amount.add(other.amount), currency: self.currency });
        }
        
        fn sub(self, other: Money) -> core.Result<Money, string> {
            if self.currency != other.currency {
                return core.Err("Currency mismatch");
            }
            return core.Ok(Money { amount: self.amount.sub(other.amount), currency: self.currency });
        }
    }
}
```

### ২. QueryBuilder থেকে সরাসরি SQLite-এ SQL জেনারেশন
এখানে দুটি অংশ: ১. Nilang-এর কোড যা SQL তৈরি করবে, ২. Go-এর কোড যা সেই SQL চালিয়ে ডেটা ফেরত দেবে।

#### ক. Nilang QueryBuilder (`std/db/query_builder.nil`)
```nilang
// std/db/query_builder.nil
namespace db {
    struct QueryBuilder {
        let table: string;
        let select_cols: List<string>;
        let where_clauses: List<string>;
        let where_params: List<any>;
        let limit_val: int;
        let order_by: string;
    }

    impl QueryBuilder {
        fn new(table: string) -> QueryBuilder {
            return QueryBuilder {
                table: table,
                select_cols: ["*"],
                where_clauses: [],
                where_params: [],
                limit_val: 0,
                order_by: ""
            };
        }

        fn select(self, cols: List<string>) -> QueryBuilder {
            self.select_cols = cols;
            return self;
        }

        fn where(self, col: string, op: string, val: any) -> QueryBuilder {
            self.where_clauses.push("\(col) \(op) ?");
            self.where_params.push(val);
            return self;
        }

        fn limit(self, n: int) -> QueryBuilder {
            self.limit_val = n;
            return self;
        }

        fn order(self, col: string, desc: bool) -> QueryBuilder {
            let dir = if desc { "DESC" } else { "ASC" };
            self.order_by = "\(col) \(dir)";
            return self;
        }

        // SQL স্ট্রিং তৈরি করার লজিক
        fn build_sql(self) -> string {
            let sql = "SELECT \(string.join(self.select_cols, ", ")) FROM \(self.table)";
            
            if self.where_clauses.length() > 0 {
                sql = sql + " WHERE " + string.join(self.where_clauses, " AND ");
            }
            
            if self.order_by != "" {
                sql = sql + " ORDER BY " + self.order_by;
            }
            
            if self.limit_val > 0 {
                sql = sql + " LIMIT \(self.limit_val)";
            }
            
            return sql;
        }

        // VM Bridge কল করা
        async fn execute(self, conn: Connection) -> core.Result<List<Map<string, any>>, string> {
            let sql = self.build_sql();
            return vm_bridge.exec_query(conn.dsn, sql, self.where_params);
        }
    }
}
```

#### খ. Go Backend Bridge (`runtime/vm/bridge_sqlite.go`)
এই কোডটি আপনার রিপোজিটরির Go রানটাইমে থাকবে।
```go
package vm

import (
    "database/sql"
    _ "github.com/mattn/go-sqlite3"
)

// Nilang থেকে আসা কোয়েরি এক্সিকিউট করার ফাংশন
func ExecQuery(dsn string, query string, args []interface{}) ([]map[string]interface{}, error) {
    db, err := sql.Open("sqlite3", dsn)
    if err != nil {
        return nil, err
    }
    defer db.Close()

    rows, err := db.Query(query, args...)
    if err != nil {
        return nil, err
    }
    defer rows.Close()

    columns, _ := rows.Columns()
    results := make([]map[string]interface{}, 0)

    for rows.Next() {
        values := make([]interface{}, len(columns))
        valuePtrs := make([]interface{}, len(columns))
        for i := range values {
            valuePtrs[i] = &values[i]
        }
        rows.Scan(valuePtrs...)

        rowMap := make(map[string]interface{})
        for i, colName := range columns {
            val := values[i]
            if b, ok := val.([]byte); ok {
                rowMap[colName] = string(b)
            } else {
                rowMap[colName] = val
            }
        }
        results = append(results, rowMap)
    }
    return results, nil
}
```

### ৩. HTTP Server / Router তৈরির নেটিভ লাইব্রেরি
একটি ফাস্ট এবং মডার্ন HTTP সার্ভার চালানোর জন্য Nilang-এর রাউটার এবং Go-এর `net/http` এর ইন্টিগ্রেশন।

#### ক. Nilang Router (`std/net/http/router.nil`)
```nilang
// std/net/http/router.nil
namespace net::http {
    struct Request {
        let method: string;
        let path: string;
        let headers: Map<string, string>;
        let body: string;
    }

    struct Response {
        let status: int;
        let headers: Map<string, string>;
        let body: string;
    }

    impl Response {
        fn json(status: int, data: any) -> Response {
            return Response {
                status: status,
                headers: {"Content-Type": "application/json"},
                body: json.stringify(data)
            };
        }
        
        fn text(status: int, data: string) -> Response {
            return Response {
                status: status,
                headers: {"Content-Type": "text/plain"},
                body: data
            };
        }
    }

    type Handler = fn(Request) -> Response;

    struct Router {
        let routes: Map<string, Handler>;
    }

    impl Router {
        fn new() -> Router {
            return Router { routes: {} };
        }

        fn get(self, path: string, handler: Handler) {
            self.routes.set("GET:" + path, handler);
        }

        fn post(self, path: string, handler: Handler) {
            self.routes.set("POST:" + path, handler);
        }

        fn handle(self, req: Request) -> Response {
            let key = "\(req.method):\(req.path)";
            if self.routes.has(key) {
                let handler = self.routes.get(key).unwrap();
                return handler(req);
            }
            return Response.text(404, "Not Found");
        }
    }

    fn serve(port: int, router: Router) {
        // Native Go Server Start
        vm_bridge.start_http_server(port, router);
    }
}
```

func (nr *NilangRouter) ServeHTTP(w http.ResponseWriter, r *http.Request) {

```nil
const E = 2.718281828459045;

let abs = fn(value) {
    if (value < 0) {
        return -value;
    }
    return value;
};

let min = fn(a, b) {
    if (a < b) {
        return a;
    }
    return b;
};

let max = fn(a, b) {
    if (a > b) {
        return a;
    }
    return b;
};

let clamp = fn(value, low, high) {
    if (value < low) {
        return low;
    }

    if (value > high) {
        return high;
    }

    return value;
};

let sign = fn(value) {
    if (value < 0) {
        return -1;
    }

    if (value > 0) {
        return 1;
    }

    return 0;
};

let isEven = fn(value) {
    return value % 2 == 0;
};

let isOdd = fn(value) {
    return value % 2 != 0;
};

let gcd = fn(a, b) {
    let x = abs(a);
    let y = abs(b);

    while (y != 0) {
        let remainder = x % y;
        x = y;
        y = remainder;
    }

    return x;
};

let lcm = fn(a, b) {
    if (a == 0 || b == 0) {
        return 0;
    }

    return abs((a / gcd(a, b)) * b);
};

let factorial = fn(n) {
    if (n < 0) {
        return null;
    }

    let result = 1;
    let i = 2;

    while (i <= n) {
        result = result * i;
        i = i + 1;
    }

    return result;
};

let powInt = fn(base, exponent) {
    if (exponent < 0) {
        return 1 / powInt(base, -exponent);
    }

    let result = 1;
    let n = exponent;

    while (n > 0) {
        if (n % 2 == 1) {
            result = result * base;
        }

        base = base * base;
        n = n / 2;
    }

    return result;
};

let fibonacci = fn(n) {
    if (n < 0) {
        return null;
    }

    if (n == 0) {
        return 0;
    }

    if (n == 1) {
        return 1;
    }

    let a = 0;
    let b = 1;
    let i = 2;

    while (i <= n) {
        let next = a + b;
        a = b;
        b = next;
        i = i + 1;
    }

    return b;
};

let average = fn(values) {
    if (len(values) == 0) {
        return null;
    }

    let total = 0;
    let i = 0;

    while (i < len(values)) {
        total = total + values[i];
        i = i + 1;
    }

    return total / len(values);
};
```

```markdown
# Nilang Standard Library

The Nilang standard library is divided into two layers.

## Core

The core standard library must remain deterministic and backend-independent.

- `std/core`
- `std/math`
- `std/strings`
- `std/collections`
- `std/testing`

These modules are implemented in Nilang itself.

## Host

Host-backed modules may access the operating system or external resources.

- `std/io`
- `std/fs`
- `std/net`
- `std/time`
- `std/json`
- `std/crypto`
- `std/process`
- `std/concurrency`

Host modules must eventually be governed by Nilang effect and capability checks.

## Design rule

The standard library must not introduce new language syntax.

Language features belong in the Nilang core.
Domain/application features belong in Alap.
Operating-system features belong in the Onuron runtime/host layer.
```
